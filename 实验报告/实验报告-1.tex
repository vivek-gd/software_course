\subsection{栈模板类实现}
\begin{lstlisting}[language=C++]
template<typename T>
class Stack {
private:
    T* data;        // 存储元素数组
    int top;        // 栈顶指针
    int capacity;   // 栈容量
    
public:
    Stack(int size=10) {
        capacity = size;
        data = new T[capacity];
        top = -1;
    }
    
    void push(T val) {
        if(top >= capacity-1) resize();
        data[++top] = val;
    }
    
    T pop() {
        if(isEmpty()) throw "Stack Underflow";
        return data[top--];
    }
};
\end{lstlisting}
